});
});
const keys = Object.keys(byAngle).sort();
document.getElementById("poly-count").textContent = keys.length + " angles";
if (!keys.length) {
el.innerHTML = '<p class="empty">No angles tagged.</p>';

}
const spotlight = (polyMeta && polyMeta.spotlight_angle) || "";
el.innerHTML = keys.map(function (a) {
const items = byAngle[a].map(function (s) {
return "<li><strong>" + esc(s.id) + "</strong> — " + esc(s.headline) + "</li>";
}).join("");
const mark = a === spotlight ? " · spotlight" : "";
return (
'<div class="angle-group">' +
'<div class="angle-group-head">' + esc(a) + esc(mark) + "</div>" +
"<ul>" + items + "</ul>" +
"</div>"
);
}).join("");
}
