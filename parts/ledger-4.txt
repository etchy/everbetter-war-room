function bindExpand(root) {
root.querySelectorAll(".directive .orbit").forEach(function (orbit) {
function toggle() {
var card = orbit.closest(".directive");
var open = card.classList.toggle("open");
orbit.setAttribute("aria-expanded", open ? "true" : "false");
}
orbit.addEventListener("click", toggle);
orbit.addEventListener("keydown", function (e) {
if (e.key === "Enter" || e.key === " ") {
e.preventDefault();
toggle();
}
});
});
root.querySelectorAll("[data-gt-more]").forEach(function (btn) {
btn.addEventListener("click", function (e) {
e.stopPropagation();
var block = btn.closest(".gt-block");
if (!block) return;
var on = block.classList.toggle("open");
btn.textContent = on ? "Collapse" : "Expand";
});
});
}
function paint(data) {
var meta = data.meta || {};
document.getElementById("data-source").textContent = "LIVE";
document.getElementById("data-source").classList.add("live");
document.getElementById("updated-at").textContent = meta.updated_at
? "updated " + meta.updated_at
: "";
document.getElementById("ledger-version").textContent = meta.version
? "v" + meta.version
: "";
if (meta.dna) document.getElementById("dna-motto").textContent = meta.dna;
var list = data.directives || [];
var el = document.getElementById("ledger");
if (!list.length) {
el.innerHTML = '<p class="empty-state">No directives yet. Edit data/directives.json.</p>';
return;
}
el.innerHTML = list.map(renderDirective).join("");
bindExpand(el);
}
