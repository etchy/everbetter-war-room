"</div>" +
"</div>"
);
}
const ku = String(f.kill_or_upgrade).toLowerCase();
const kuClass = ku.indexOf("kill") >= 0 ? "ff-kill" : (ku.indexOf("upgrade") >= 0 ? "ff-upgrade" : "");

'<div class="forge-field' + (extraClass ? " " + extraClass : "") + '">' +
'<div class="ff-k">' + esc(key) + "</div>" +

"</div>"
);
}
return (
'<div class="forge-item forge-valid">' +
'<span class="forge-status ' + esc(f.status || "") + '">' + esc((f.status || "").replace(/_/g, " ")) + "</span>" +
'<div class="forge-body">' +
'<div class="forge-title">' + esc(f.title || f.id || "") + "</div>" +

field("kill_or_upgrade", f.kill_or_upgrade, kuClass) +

"</div>" +

"</div>"
);
}).join("");
}
