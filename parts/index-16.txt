function applyOpsAging(board, freshness) {
const oa = (freshness && freshness.ops_aging) || { fresh_hours: 2, aging_hours: 8 };
const freshH = Number(oa.fresh_hours) || 2;
const agingH = Number(oa.aging_hours) || 8;
const now = Date.now();
(board.active_stories || []).forEach(function (s) {
if (!s.updated_at) return;
const t = Date.parse(s.updated_at);
if (Number.isNaN(t)) return;
const hours = (now - t) / 36e5;
if (hours > agingH) {
s._opsAge = "stale";
s._opsAgeLabel = "STALE";
} else if (hours > freshH) {
s._opsAge = "aging";
s._opsAgeLabel = "AGING";
} else {
s._opsAge = "fresh";
s._opsAgeLabel = "FRESH";
}
});
}
